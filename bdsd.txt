// 4. XỬ LÝ GIAO DỊCH VPBANK: ZALO BOT + SAPO AUTO ORDER
// ============================================
function processSource_(entries, store, sourceKey) {
  const digestKey = STORE_KEYS.lastDigest;
  const topEntryKeyKey = STORE_KEYS.lastTopEntryKey;
  
  const limitedEntries = entries.slice(0, CONFIG.digestEntryLimit);
  const currentDigest = buildDigest_(limitedEntries);
  const currentTopKey = buildEntryKey_(limitedEntries[0]);
  
  const previousDigest = store.getProperty(digestKey);
  const previousTopKey = store.getProperty(topEntryKeyKey);
  const isFirstRun = !previousDigest;

  if (isFirstRun && CONFIG.bootstrapWithoutNotification) {
    store.setProperty(digestKey, currentDigest);
    store.setProperty(topEntryKeyKey, currentTopKey);
    Logger.log("[" + sourceKey + "] Bootstrap complete.");
    return;
  }

  if (currentDigest === previousDigest) {
    Logger.log("ℹ️ VPBank: Không có giao dịch phát sinh mới.");
    return;
  }

  const newEntries = getNewEntries_(limitedEntries, previousTopKey);
  if (newEntries.length > 0) {
    const currentBalance = parseFloat(store.getProperty(STORE_KEYS.balance) || "0");
    let runningBalance = currentBalance;
    
    const processedEntries = newEntries.slice().reverse().map(function(e) {
      const val = e.amountValue;
      if (e.sign === "+") {
        runningBalance += val;
      } else {
        runningBalance -= val;
      }
      e.balanceAfter = runningBalance;
      return e;
    });

    // 1. XỬ LÝ TỰ ĐỘNG TẠO ĐƠN SAPO NẾU LÀ TIỀN VÀO HỢP LỆ (20k, 50k, 500k) TRƯỚC KHI GỬI ZALO
    const validAmounts = [20000, 50000, 500000];
    newEntries.forEach(function(entry) {
      if (entry.sign === "+" && validAmounts.includes(entry.amountValue)) {
        const orderResult = processAutoOrderForEntry_(entry);
        if (orderResult && orderResult.success) {
          entry.orderStatus = { isAutoScan: true, success: true, customerName: orderResult.customerName };
        } else {
          entry.orderStatus = { isAutoScan: true, success: false };
        }
      }
    });

    // 2. GỬI THÔNG BÁO VỀ ZALO BOT (KÈM TRẠNG THÁI TẠO ĐƠN)
    sendFundNotification_(processedEntries.reverse());
    store.setProperty(STORE_KEYS.balance, runningBalance.toString());
  }

  store.setProperty(digestKey, currentDigest);
  store.setProperty(topEntryKeyKey, currentTopKey);
}

// TỰ ĐỘNG TẠO ĐƠN SAPO (CHẠY 1 LẦN DUY NHẤT)
// ----------------------------------
function processAutoOrderForEntry_(entry) {
  Logger.log("⚡ Xử lý tạo đơn cho giao dịch: " + entry.amount + " | ND: " + entry.name);
  
  // So khớp thuần theo Tên Facebook của khách hàng
  const record = findPendingCustomerByName_(entry.name);
  if (!record) {
    Logger.log("⚠️ Chưa tìm thấy khách nào khớp với Tên FB trong nội dung: " + entry.name);
    return { success: false, reason: "Không tìm thấy khách" };
  }

  const token = getSapoToken_();
  if (!token) {
    Logger.log("❌ Không có Token Sapo! Mở trình duyệt PC để extension tự động đồng bộ token.");
    return { success: false, reason: "Thiếu Token" };
  }

  try {
    Logger.log("🛒 Tạo đơn Sapo cho khách: " + record.customerName + " (Gói: " + entry.amountValue + "đ, Page: " + record.pageId + ")");
    const orderResult = createSapoOrder_({
      conversationId: record.convId,
      pageId: record.pageId,
      fbPageId: record.fbPageId,
      customerName: record.customerName,
      amount: entry.amountValue,
      token: token,
      tenant: CONFIG.sapoTenant
    });

    if (orderResult && orderResult.success) {
      updateTransferRecordStatus_(record.rowNumber, "ORDERED", entry.amountValue, orderResult.orderId);
      Logger.log("✅ ĐÃ TẠO ĐƠN THÀNH CÔNG! Mã đơn Sapo: #" + orderResult.orderId);
      return { success: true, orderId: orderResult.orderId, customerName: record.customerName };
    } else {
      Logger.log("❌ Lỗi tạo đơn Sapo: " + (orderResult ? orderResult.error : "Không rõ nguyên nhân"));
      return { success: false, reason: orderResult ? orderResult.error : "Lỗi API Sapo" };
    }
  } catch (err) {
    Logger.log("❌ Ngoại lệ khi tạo đơn: " + err.message);
    return { success: false, reason: err.message };
  }
}

    if (cRes.getResponseCode() === 200) {

      if (res.getResponseCode() === 200) {

      }
    } catch(errPage) {}
  });
}

function removeVietnameseTones_(str) {
  if (!str) return "";
  str = String(str);
  // 1. Chuẩn hóa Unicode NFD để bóc tách toàn bộ dấu tổ hợp tiếng Việt
  try {
    str = str.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
  } catch (e) {}
  // 2. Thay thế bảng mã ký tự tiếng Việt có dấu dựng sẵn (NFC)
  str = str.replace(/à|á|ạ|ả|ã|â|ầ|ấ|ậ|ẩ|ẫ|ă|ằ|ắ|ặ|ẳ|ẵ/g, "a");
  str = str.replace(/è|é|ẹ|ẻ|ẽ|ê|ề|ế|ệ|ể|ễ/g, "e");
  str = str.replace(/ì|í|ị|ỉ|ĩ/g, "i");
  str = str.replace(/ò|ó|ọ|ỏ|õ|ô|ồ|ố|ộ|ổ|ỗ|ơ|ờ|ớ|ợ|ở|ỡ/g, "o");
  str = str.replace(/ù|ú|ụ|ủ|ũ|ư|ừ|ứ|ự|ử|ữ/g, "u");
  str = str.replace(/ỳ|ý|ỵ|ỷ|ỹ/g, "y");
  str = str.replace(/đ/g, "d");
  str = str.replace(/À|Á|Ạ|Ả|Ã|Â|Ầ|Ấ|Ậ|Ẩ|Ẫ|Ă|Ằ|Ắ|Ặ|Ẳ|Ẵ/g, "A");
  str = str.replace(/È|É|Ẹ|Ẻ|Ẽ|Ê|Ề|Ế|Ệ|Ể|Ễ/g, "E");
  str = str.replace(/Ì|Í|Ị|Ỉ|Ĩ/g, "I");
  str = str.replace(/Ò|Ó|Ọ|Ỏ|Õ|Ô|Ồ|Ố|Ộ|Ổ|Ỗ|Ơ|Ờ|Ớ|Ợ|Ở|Ỡ/g, "O");
  str = str.replace(/Ù|Ú|Ụ|Ủ|Ũ|Ư|Ừ|Ứ|Ự|Ử|Ữ/g, "U");
  str = str.replace(/Ỳ|Ý|Ṇ|Ỷ|Ỹ/g, "Y");
  str = str.replace(/Đ/g, "D");
  return str.replace(/[^a-zA-Z0-9\s]/g, " ").replace(/\s+/g, " ").trim().toUpperCase();
}

function extractTransferRemark_(bankText) {
  if (!bankText) return "";
  let text = removeVietnameseTones_(bankText);
  
  // 1. Cắt bỏ tiền tố hệ thống ngân hàng (VPBank, Napas, Trace, STK, Lời nhắn...)
  // Nếu có chữ "ND:" hoặc "ND QR" hoặc "NOI DUNG" hoặc "LOI NHAN" -> Lấy phần nội dung thực sự phía sau
  const ndIndex = text.search(/\b(?:ND|NOI DUNG|ND QR|NOI DUNG QR|LOI NHAN|MESSAGE)[\s:_-]+/i);
  if (ndIndex !== -1) {
    text = text.substring(ndIndex).replace(/^(?:ND|NOI DUNG|ND QR|NOI DUNG QR|LOI NHAN|MESSAGE)[\s:_-]+/i, "");
  } else {
    // Nếu không có tiền tố rõ ràng -> xóa các từ khóa hệ thống ngân hàng
    text = text.replace(/\b(?:VPBANK|VPB|TECHCOMBANK|TCB|VIETCOMBANK|VCB|MBBANK|MBVCB|BIDV|AGRIBANK|ACB|TPB|NAPAS|TRACE|EBANK|IB\w+|FT\w+|STK|NHAN TU|NHAN TIEN)\b[\s\d]*/gi, " ");
  }

  // 2. Cắt bỏ tiền tố nhận diện phụ ở đầu (FB, FACEBOOK, ACC, TK, NICK...)
  text = text.replace(/^(?:FB|FACEBOOK|ACC|TK|NICK|QR)\s+/i, "");

  // 3. Cắt bỏ hậu tố phụ trợ ở đuôi (CHUYEN TIEN, CHUYEN KHOAN, CHUYEN, CK, THANH TOAN, TT...)
  text = text.replace(/(?:\s+(?:CHUYEN TIEN|CHUYEN KHOAN|CHUYEN|CK|THANH TOAN|TT|MUA|XEM QUE|XEM|QUE|KINH DICH|TU VI))+\s*$/i, "");

  // 4. Làm sạch khoảng trắng thừa và ký tự đặc biệt
  text = text.replace(/[^A-Z0-9\s]/g, " ").replace(/\s+/g, " ").trim();
  return text;
}

function findPendingCustomerByName_(bankText) {
  const ss = getSpreadsheet_();
  if (!ss) return null;
  const sheet = ss.getSheetByName("Transfers");
  if (!sheet) return null;

  // Trích xuất nội dung thuần của người chuyển khoản (đã lọc sạch rác ngân hàng)
  const cleanRemark = extractTransferRemark_(bankText);
  const compactRemark = cleanRemark.replace(/\s+/g, "");
  
  // Yêu cầu nội dung CK:
  // 1. Có từ 2 từ trở lên (mỗi từ >= 2 ký tự), HOẶC
  // 2. Viết liền 1 từ NHƯNG độ dài >= 5 ký tự (khách viết liền cả họ và tên như "phamquantran", "nguyenoanh", "manche")
  // Tránh trường hợp chỉ có 1 từ ngắn đơn lẻ (VD: "HIEN", "AN", "HA" < 5 ký tự) gây khớp nhầm
  const remarkWords = cleanRemark.split(/\s+/).filter(w => w.length >= 2);
  if (remarkWords.length < 2 && compactRemark.length < 5) {
    Logger.log("🛑 Nội dung CK quá ngắn hoặc chỉ có 1 từ đơn lẻ ('" + cleanRemark + "') -> Không đủ cơ sở định danh an toàn. BỎ QUA để tránh tạo nhầm.");
    return null;
  }

  const data = sheet.getDataRange().getValues();
  const pendingCandidates = [];
  const orderedCandidates = [];

  for (let i = data.length - 1; i >= 1; i--) {
    const custName = String(data[i][4] || "").trim();
    const status = String(data[i][5] || "").trim().toUpperCase();
    if (custName) {
      const cleanName = removeVietnameseTones_(custName);
      const candWords = cleanName.split(/\s+/).filter(w => w.length >= 2);
      // Bắt buộc tên khách FB cũng phải có ít nhất 2 từ (độ dài >= 4) để đủ cơ sở định danh
      if (candWords.length >= 2 && cleanName.length >= 4) {
        const reversedWords = candWords.slice().reverse();
        const item = {
          rowNumber: i + 1,
          convId: data[i][1],
          pageId: data[i][2],
          fbPageId: data[i][3],
          customerName: custName,
          cleanName: cleanName,
          candWords: candWords,
          reversedWords: reversedWords,
          compactName: cleanName.replace(/\s+/g, ""),
          compactReversed: reversedWords.join(""),
          tagTime: data[i][8] || "",
          status: status
        };
        if (status === "PENDING") {
          pendingCandidates.push(item);
        } else {
          orderedCandidates.push(item);
        }
      }
    }
  }

  function matchCandidateStrict_(list) {
    if (list.length === 0) return null;

    let bestCand = null;
    let bestScore = 0;
    let bestMatchType = "";

    for (let c = 0; c < list.length; c++) {
      const cand = list[c];
      let candScore = 0;
      let matchType = "";

      const normalWords = cand.candWords;
      const reversedWords = cand.reversedWords;
      const reversedCleanName = reversedWords.join(" ");

      // QUY TẮC 1: Khớp NGUYÊN CỤM TÊN liên tiếp (Thuận hoặc Đảo kiểu Tây/Việt)
      // Ví dụ: FB là "Nguyễn Mai Vương" hoặc "Se Rena" -> Tìm đúng cụm "NGUYEN MAI VUONG" hoặc "SE RENA"
      const exactPatternNormal = new RegExp("(?:^|[^A-Z0-9])" + escapeRegex_(cand.cleanName) + "(?:$|[^A-Z0-9])");
      const exactPatternRev = new RegExp("(?:^|[^A-Z0-9])" + escapeRegex_(reversedCleanName) + "(?:$|[^A-Z0-9])");

      if (exactPatternNormal.test(" " + cleanRemark + " ")) {
        candScore = 100 + normalWords.length * 10;
        matchType = "Khớp nguyên cụm tên ('" + cand.cleanName + "')";
      } else if (exactPatternRev.test(" " + cleanRemark + " ")) {
        candScore = 98 + normalWords.length * 10;
        matchType = "Khớp nguyên cụm tên đảo ('" + reversedCleanName + "')";
      }
      // QUY TẮC 2: Khớp dính liền không dấu toàn bộ Họ và Tên (Hỗ trợ cả tên dính liền như Thuytrang Nguyen <-> THUY TRANG NGUYEN)
      // Ví dụ: FB "Thuytrang Nguyen" (THUYTRANGNGUYEN) khớp hoàn hảo với CK "THUY TRANG NGUYEN" (THUYTRANGNGUYEN)
      else if (cand.compactName.length >= 5 && (compactRemark.includes(cand.compactName) || (cand.compactReversed && compactRemark.includes(cand.compactReversed)))) {
        candScore = 90 + cand.compactName.length;
        matchType = "Khớp viết liền toàn bộ Họ & Tên ('" + cand.compactName + "')";
      }
      // QUY TẮC 3: Khớp đảo Họ - Tên LIỀN KỀ (Chỉ cho phép 2 từ đứng sát nhau)
      // Ví dụ: FB là "Tha Bùi" -> trong nội dung phải có cụm "BUI THA" đứng sát nhau
      else if (normalWords.length === 2) {
        const revPhrase = normalWords[1] + " " + normalWords[0]; // "BUI" + " " + "THA"
        const revPattern = new RegExp("(?:^|[^A-Z0-9])" + escapeRegex_(revPhrase) + "(?:$|[^A-Z0-9])");
        if (revPattern.test(" " + cleanRemark + " ")) {
          candScore = 95;
          matchType = "Khớp đảo Họ-Tên liền kề ('" + revPhrase + "')";
        }
      }

      // QUY TẮC 4: Khớp Họ & Tên chính theo thứ tự (cho phép thêm 1-2 tên đệm ở giữa)
      // Tự động kiểm tra cả chiều Việt (Nguyễn Hồng Nhung) lẫn chiều Tây (Trang Thu Nguyễn / Nhan Dinh)
      // Bắt buộc: Từ ĐẦU TIÊN (Họ) và từ CUỐI CÙNG (Tên chính) phải khớp đúng tuyệt đối
      if (candScore === 0 || candScore < 100) {
        const testWordVariants = [normalWords, reversedWords];
        for (let v = 0; v < testWordVariants.length; v++) {
          const fbWords = testWordVariants[v];
          const ckWords = remarkWords;
          const extraWords = ckWords.length - fbWords.length;
          
          if (extraWords >= 1 && extraWords <= 2) {
            const sameFirst = (fbWords[0] === ckWords[0]);
            const sameLast = (fbWords[fbWords.length - 1] === ckWords[ckWords.length - 1]);
            
            if (sameFirst && sameLast) {
              let ckIdx = 0;
              let matchedCount = 0;
              for (let f = 0; f < fbWords.length; f++) {
                while (ckIdx < ckWords.length) {
                  if (ckWords[ckIdx] === fbWords[f]) {
                    matchedCount++;
                    ckIdx++;
                    break;
                  }
                  ckIdx++;
                }
              }
              if (matchedCount === fbWords.length) {
                // Điểm = 80 + số từ khớp * 15 (Người khớp 3 từ như Nguyễn Hồng Nhung = 125 điểm > người chỉ khớp 2 từ như Hồng Nhung = 120 điểm)
                const score = 80 + matchedCount * 15;
                if (score > candScore) {
                  candScore = score;
                  matchType = "Khớp Họ & Tên chính theo thứ tự ('" + fbWords.join(" ") + "' trong '" + cleanRemark + "')";
                }
              }
            }
          }
        }
      }

      // Lưu lại ứng viên có điểm số cao nhất
      if (candScore > bestScore) {
        bestScore = candScore;
        bestCand = cand;
        bestMatchType = matchType;
      }
    }

    if (bestCand && bestScore > 0) {
      Logger.log("✅ " + bestMatchType + " (Điểm: " + bestScore + ") -> Khách: '" + bestCand.customerName + "' (Dòng " + bestCand.rowNumber + ")");
      return bestCand;
    }

    return null;
  }

  // 1. Ưu tiên tìm trong nhóm PENDING (khách vừa gắn nhãn)
  const matchedPending = matchCandidateStrict_(pendingCandidates);
  if (matchedPending) {
    Logger.log("🎯 Khớp chính xác khách PENDING: '" + matchedPending.customerName + "' (Dòng " + matchedPending.rowNumber + ")");
    return matchedPending;
  }

  // 2. Tìm trong nhóm khách cũ ORDERED (khách quen tự chuyển khoản lại)
  const matchedOrdered = matchCandidateStrict_(orderedCandidates);
  if (matchedOrdered) {
    Logger.log("🎯 Khớp chính xác khách quen chuyển khoản lại: '" + matchedOrdered.customerName + "' (Dòng " + matchedOrdered.rowNumber + ")");
    return matchedOrdered;
  }

  Logger.log("🛑 Không khớp được khách nào với nội dung sau khi lọc: '" + cleanRemark + "' (Thà bỏ qua để an toàn)");
  return null;
}

function escapeRegex_(str) {
  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

function getSpreadsheet_() {
  try {
    const active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active;
  } catch (e) {}

  try {
    if (CONFIG.spreadsheetId && !CONFIG.spreadsheetId.includes("DIEN_ID")) {
      return SpreadsheetApp.openById(CONFIG.spreadsheetId);
    }
  } catch (e) {}

  return null;
}

function registerPendingTransfer_(data) {
  const ss = getSpreadsheet_();
  if (!ss) return;
  const sheet = ss.getSheetByName("Transfers");
  if (!sheet) return;

  const rows = sheet.getDataRange().getValues();
  let foundRow = -1;
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][1]).trim() === String(data.convId).trim()) {
      foundRow = i + 1;
      break;
    }
  }

  const nowStr = new Date().toLocaleString("vi-VN");

  if (foundRow > 0) {
    sheet.getRange(foundRow, 3).setValue(data.pageId);
    sheet.getRange(foundRow, 4).setValue(data.fbPageId || "");
    sheet.getRange(foundRow, 5).setValue(data.customerName || "Khách Hàng");
    sheet.getRange(foundRow, 6).setValue("PENDING");
    sheet.getRange(foundRow, 9).setValue(nowStr);
    Logger.log("✅ Đã cập nhật dòng " + foundRow + " cho khách: " + data.customerName);
  } else {
    sheet.appendRow([
      rows.length, // STT
      data.convId,
      data.pageId,
      data.fbPageId || "",
      data.customerName || "Khách Hàng",
      "PENDING",
      "",
      "",
      nowStr,
      ""
    ]);
    Logger.log("✅ Đã thêm mới dòng cho khách: " + data.customerName);
  }
}

function findPendingRecordByConvId_(convId) {
  const ss = getSpreadsheet_();
  if (!ss) return null;
  const sheet = ss.getSheetByName("Transfers");
  if (!sheet) return null;

  const data = sheet.getDataRange().getValues();
  for (let i = data.length - 1; i >= 1; i--) {
    const rowConvId = String(data[i][1]).trim();
    const status = String(data[i][5]).trim().toUpperCase();
    if (rowConvId === convId && status === "PENDING") {
      return { rowNumber: i + 1, convId: data[i][1] };
    }
  }
  return null;
}

function updateTransferRecordStatus_(rowNumber, status, amount, orderId) {
  const ss = getSpreadsheet_();
  if (!ss) return;
  const sheet = ss.getSheetByName("Transfers");
  if (!sheet) return;

  const currentAmount = String(sheet.getRange(rowNumber, 7).getValue() || "").trim();
  const currentOrderId = String(sheet.getRange(rowNumber, 8).getValue() || "").trim();
  const currentOrderTime = String(sheet.getRange(rowNumber, 10).getValue() || "").trim();

  const newAmount = currentAmount ? (currentAmount + " | " + amount) : amount;
  const newOrderId = currentOrderId ? (currentOrderId + ", " + orderId) : orderId;
  const newTime = currentOrderTime ? (currentOrderTime + " | " + new Date().toLocaleString("vi-VN")) : new Date().toLocaleString("vi-VN");

  sheet.getRange(rowNumber, 6).setValue(status);
  sheet.getRange(rowNumber, 7).setValue(newAmount);
  sheet.getRange(rowNumber, 8).setValue(newOrderId);
  sheet.getRange(rowNumber, 10).setValue(newTime);
}

function getSapoToken_() {
  const store = PropertiesService.getScriptProperties();
  let token = store.getProperty(STORE_KEYS.sapoToken);
  if (token) return token;

  const ss = getSpreadsheet_();
  if (ss) {
    const configSheet = ss.getSheetByName("Config");
    if (configSheet) {
      const data = configSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (data[i][0] === "sapo_token" && data[i][1]) {
          return String(data[i][1]).trim();
        }
      }
    }
  }
  return "";
}

function saveSapoToken_(token) {
  if (!token) return;
  PropertiesService.getScriptProperties().setProperty(STORE_KEYS.sapoToken, token);
  
  const ss = getSpreadsheet_();
  if (ss) {
    const configSheet = ss.getSheetByName("Config");
    if (configSheet) {
      const data = configSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (data[i][0] === "sapo_token") {
          configSheet.getRange(i + 1, 2).setValue(token);
          configSheet.getRange(i + 1, 3).setValue(new Date().toLocaleString("vi-VN"));
          return;
        }
      }
    }
  }
}

function sendFundNotification_(newEntries) {
  if (!CONFIG.zaloChatId || newEntries.length === 0) return;

  const lines = newEntries.slice().reverse().map(function (e) {
    const icon = e.sign === "+" ? "🟢" : "🔴";
    let textBlock = icon + " " + e.sign + e.amount;
    
    // Nếu là giao dịch thuộc diện quét tạo đơn (20k, 50k, 500k)
    if (e.orderStatus && e.orderStatus.isAutoScan) {
      if (e.orderStatus.success && e.orderStatus.customerName) {
        textBlock += "\n✅ " + e.orderStatus.customerName;
      } else {
        textBlock += "\n❌ Chưa tạo đơn";
      }
    }

    textBlock += "\n💰 Số dư: " + formatCurrency_(e.balanceAfter) + "đ\n" +
                 "⏰ Thời gian: " + e.dateText + "\n" +
                 "📝 Nội dung: " + e.name;
    return textBlock;
  });

  var text = lines.join("\n\n------------------\n\n");
  sendZaloMessage_(CONFIG.zaloBotToken, CONFIG.zaloChatId, text);
}

function sendZaloMessage_(token, chatId, text) {
  var url = "https://bot-api.zaloplatforms.com/bot" + token + "/sendMessage";
  var payload = { "chat_id": String(chatId), "text": text };

  var response = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });
  
  if (response.getResponseCode() >= 300) {
    Logger.log("Lỗi Zalo: " + response.getContentText());
  }
}

// Webhook xử lý cả Zalo Webhook lẫn Token Sync từ Extension
function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) return ContentService.createTextOutput("No data");
    
    let data = {};
    try { data = JSON.parse(e.postData.contents); } catch(err) {}

    // 1. Nếu là lệnh từ Extension (Đồng bộ Token Sapo)
    if (data.action === "sync_token" && data.token) {
      saveSapoToken_(data.token);
      return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Token updated" })).setMimeType(ContentService.MimeType.JSON);
    }

    // 1b. Đồng bộ Page IDs từ Extension
    if (data.action === "sync_pages" && Array.isArray(data.page_ids)) {
      const store = PropertiesService.getScriptProperties();
      let pids = JSON.parse(store.getProperty("SAPO_PAGE_IDS") || "[]");
      data.page_ids.forEach(function(pid) {
        if (pid && !pids.includes(pid)) pids.push(pid);
      });
      store.setProperty("SAPO_PAGE_IDS", JSON.stringify(pids));
      Logger.log("✅ Đã lưu " + pids.length + " Page IDs từ Extension: " + pids.join(", "));
      return ContentService.createTextOutput(JSON.stringify({ success: true, count: pids.length })).setMimeType(ContentService.MimeType.JSON);
    }

    // 2. Nếu là Webhook từ Zalo Bot
    var userId = "";
    if (data.message && data.message.chat && data.message.chat.id) {
      userId = String(data.message.chat.id);
    } else if (data.sender && data.sender.id) {
      userId = String(data.sender.id);
    }
    
    if (userId !== "") {
      sendZaloMessage_(CONFIG.zaloBotToken, userId, "✅ Chat ID của bạn là: " + userId + "\n\nHãy copy dãy số này và dán vào phần zaloChatId ở dòng 23 trong code nhé!");
    }
    
    return ContentService.createTextOutput("ok");
  } catch(err) {
    return ContentService.createTextOutput("error: " + err.message);
  }
}

function fetchVPBankEntries_() {
  if (!CONFIG.vpbankUrl) return [];

  const response = UrlFetchApp.fetch(CONFIG.vpbankUrl, {
    method: "get",
    headers: {
      "accept": "application/json",
      "authkey": CONFIG.vpbankAuthKey,
      "cookie": CONFIG.vpbankCookie,
      "accountnumber": CONFIG.vpbankAccountNumber,
      "keyshare": CONFIG.vpbankKeyshare,
      "pinshare": CONFIG.vpbankPinshare,
      "referer": "https://neo.vpbank.com.vn/notification-list.html",
      "user-agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/146.0.0.0 Safari/537.36",
      "x-requested-with": "XMLHttpRequest"
    },
    muteHttpExceptions: true
  });

  if (response.getResponseCode() >= 300) {
    throw new Error("VPBank HTTP " + response.getResponseCode());
  }

  const json = JSON.parse(response.getContentText());
  if (!json || !json.d || !json.d.Message) return [];

  const messageObj = JSON.parse(json.d.Message);
  const accountKey = Object.keys(messageObj)[0];
  if (!accountKey) return [];

  const rawEntries = messageObj[accountKey];
  const entries = [];

  for (let i = 0; i < rawEntries.length; i++) {
    const parts = rawEntries[i].split("|");
    if (parts.length >= 5) {
      const datePart = parts[0].replace("VPB:", "").trim();
      let changeStr = parts[2].replace("VND", "").replace(/,/g, "").trim();
      let sign = "+";
      if (changeStr.indexOf("-") === 0) {
        sign = "-";
        changeStr = changeStr.substring(1);
      }
      
      let amountFormatted = changeStr.replace(/\B(?=(\d{3})+(?!\d))/g, ".");
      const desc = parts[4].trim();

      entries.push({
        source: "VPBank",
        sign: sign,
        amount: amountFormatted + "đ",
        amountValue: parseFloat(changeStr),
        name: desc,
        dateText: datePart
      });
    }
  }
  return entries;
}

function getNewEntries_(entries, previousTopEntryKey) {
  if (!previousTopEntryKey) return entries; 
  const newEntries = [];
  for (let i = 0; i < entries.length; i += 1) {
    if (buildEntryKey_(entries[i]) === previousTopEntryKey) break;
    newEntries.push(entries[i]);
  }
  return newEntries;
}

function buildDigest_(entries) {
  var payload = entries.map(buildEntryKey_).join("\n");
  var digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, payload);
  return Utilities.base64EncodeWebSafe(digest);
}

function buildEntryKey_(entry) {
  return [entry.sign, entry.amount, entry.name, entry.dateText].join("|");
}

function formatCurrency_(amount) {
  if (isNaN(amount)) return "0";
  return amount.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ".");
}

  const store = PropertiesService.getScriptProperties();
