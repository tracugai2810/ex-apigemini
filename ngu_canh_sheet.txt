function getSpreadsheetSafe_() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

function getContextSheet_(ss) {
  return ss.getSheetByName("Ngữ cảnh Extension")
    || ss.getSheetByName("Ngữ cảnh")
    || ss.getSheetByName("Context")
    || ss.getSheets()[0];
}

function findRowsByConvId_(sheet, convId) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  const ranges = sheet
    .getRange(2, 1, lastRow - 1, 1)
    .createTextFinder(convId)
    .matchEntireCell(true)
    .findAll();
  return ranges.map(r => r.getRow()).sort((a, b) => a - b);
}

function readConvProfileFromSheet_(convId) {
  const cid = String(convId || "").trim();
  if (!cid || cid === "default") return { summaries: [], updatedAt: "" };
  const ss = getSpreadsheetSafe_();
  const sheet = getContextSheet_(ss);
  const matchedRows = findRowsByConvId_(sheet, cid);
  if (matchedRows.length === 0) return { summaries: [], updatedAt: "" };

  const targetRow = matchedRows[matchedRows.length - 1];
  const rowValues = sheet.getRange(targetRow, 1, 1, 4).getValues()[0];
  let summaries = [];
  const rowTime = rowValues[2] ? String(rowValues[2]) : "";
  try {
    summaries = JSON.parse(rowValues[1]);
    if (!Array.isArray(summaries)) summaries = [{ text: String(rowValues[1]), time: rowTime }];
  } catch (err) {
    if (rowValues[1]) summaries = [{ text: String(rowValues[1]), time: rowTime }];
  }
  return { summaries: summaries, updatedAt: rowTime };
}

function saveConvProfileToSheet_(convId, customerName, profileText) {
  const cid = String(convId || "").trim();
  let text = String(profileText || "").trim();
  if (!cid || cid === "default" || !text) {
    throw new Error("Missing convId or data");
  }
  // Giới hạn an toàn 45.000 ký tự (trần 50.000 ký tự / ô Google Sheet)
  if (text.length > 45000) {
    text = text.slice(-45000);
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = getSpreadsheetSafe_();
    const sheet = getContextSheet_(ss);
    const nowStr = Utilities.formatDate(new Date(), "Asia/Ho_Chi_Minh", "dd/MM/yyyy HH:mm:ss");
    const finalJsonData = JSON.stringify([{ text: text, time: nowStr }]);
    const matchedRows = findRowsByConvId_(sheet, cid);
    const foundRow = matchedRows.length > 0 ? matchedRows[matchedRows.length - 1] : -1;

    if (foundRow > 0) {
      sheet.getRange(foundRow, 1).setValue(cid);
      sheet.getRange(foundRow, 2).setValue(finalJsonData);
      sheet.getRange(foundRow, 3).setValue(nowStr);
      if (customerName) sheet.getRange(foundRow, 4).setValue(customerName);
      for (let i = matchedRows.length - 2; i >= 0; i--) {
        sheet.deleteRow(matchedRows[i]);
      }
    } else {
      sheet.appendRow([cid, finalJsonData, nowStr, customerName || "Khách Hàng"]);
    }
    return nowStr;
  } finally {
    try { lock.releaseLock(); } catch (eLock) {}
  }
}

function deleteConvProfileFromSheet_(convId) {
  const cid = String(convId || "").trim();
  if (!cid || cid === "default") return;
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = getSpreadsheetSafe_();
    const sheet = getContextSheet_(ss);
    const matchedRows = findRowsByConvId_(sheet, cid);
    for (let i = matchedRows.length - 1; i >= 0; i--) {
      sheet.deleteRow(matchedRows[i]);
    }
  } finally {
    try { lock.releaseLock(); } catch (eLock) {}
  }
}

function doGet(e) {
  try {
    const params = (e && e.parameter) ? e.parameter : {};
    const action = params.action || "";

    // Nếu mở trực tiếp trên trình duyệt (không truyền action hoặc action=ui/mobile) -> Trả Web Mobile
    if (!action || action === "ui" || action === "mobile") {
      let initialStateJson = "null";
      try {
        initialStateJson = JSON.stringify(mobileGetDashboardState({}))
          .replace(/</g, "\\u003c")
          .replace(/>/g, "\\u003e")
          .replace(/\u2028/g, "\\u2028")
          .replace(/\u2029/g, "\\u2029");
      } catch (eInit) {}
      return HtmlService.createHtmlOutput(getMobileWebHtml_(initialStateJson))
        .setTitle("Kinh Dịch Mobile — Hồ Sơ & Luận Quẻ")
        .addMetaTag("viewport", "width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no")
        .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    }

    const convId = String(params.convId || "").trim();
    if (!convId || convId === "default") {
      return ContentService.createTextOutput(JSON.stringify({ success: true, data: [], updatedAt: "" })).setMimeType(ContentService.MimeType.JSON);
    }

    const ss = getSpreadsheetSafe_();
    const sheet = getContextSheet_(ss);

    // 1. ACTION GET: ĐỌC LỊCH SỬ CHỈ THEO CỘT A (convId)
    if (action === "get") {
      const res = readConvProfileFromSheet_(convId);
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        data: res.summaries,
        updatedAt: res.updatedAt
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // 2. ACTION SAVE (GET CŨ): LƯU TÓM TẮT CHỈ THEO CỘT A (convId)
    if (action === "save") {
      let customerName = String(params.customerName || "").trim();
      try {
        if (customerName.includes("%")) customerName = decodeURIComponent(customerName);
      } catch (eCust) {}

      let rawData = String(params.data || "").trim();
      try {
        if (rawData.includes("%")) rawData = decodeURIComponent(rawData);
      } catch (eData) {}

      const nowStr = new Date().toLocaleString("vi-VN");
      const matchedRows = findRowsByConvId_(sheet, convId);
      let foundRow = -1;
      let existingSummaries = [];

      if (matchedRows.length > 0) {
        foundRow = matchedRows[matchedRows.length - 1];
        const rowValues = sheet.getRange(foundRow, 1, 1, 4).getValues()[0];
        try {
          existingSummaries = JSON.parse(rowValues[1]);
          if (!Array.isArray(existingSummaries)) existingSummaries = [{ text: String(rowValues[1]), time: rowValues[2] }];
        } catch (err) {
          if (rowValues[1]) existingSummaries = [{ text: String(rowValues[1]), time: rowValues[2] }];
        }
      }

      let newSummaryObj = { text: rawData, time: nowStr };
      try {
        const parsed = JSON.parse(rawData);
        if (Array.isArray(parsed)) {
          existingSummaries = parsed;
          newSummaryObj = null;
        }
      } catch (eJson) {}

      if (newSummaryObj) {
        const lastSummary = existingSummaries[existingSummaries.length - 1];
        const lastText = typeof lastSummary === 'string' ? lastSummary : (lastSummary && lastSummary.text ? lastSummary.text : "");
        if (!lastText || lastText.trim() !== newSummaryObj.text.trim()) {
          existingSummaries.push(newSummaryObj);
          if (existingSummaries.length > 3) existingSummaries = existingSummaries.slice(-3);
        }
      }

      const finalJsonData = JSON.stringify(existingSummaries);

      if (foundRow > 0) {
        sheet.getRange(foundRow, 1).setValue(convId);
        sheet.getRange(foundRow, 2).setValue(finalJsonData);
        sheet.getRange(foundRow, 3).setValue(nowStr);
        if (customerName) sheet.getRange(foundRow, 4).setValue(customerName);
      } else {
        sheet.appendRow([
          convId,
          finalJsonData,
          nowStr,
          customerName || "Khách Hàng"
        ]);
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true, time: nowStr })).setMimeType(ContentService.MimeType.JSON);
    }

    // 3. ACTION DELETE: XÓA CHỈ THEO convId (CỘT A)
    if (action === "delete") {
      deleteConvProfileFromSheet_(convId);
      return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Invalid action" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: "No post data" })).setMimeType(ContentService.MimeType.JSON);
    }

    let body = {};
    try { body = JSON.parse(e.postData.contents); } catch (err) {}

    // 1. ACTION CŨ CỦA EXTENSION PC: LƯU ĐÈ HỒ SƠ NGỮ CẢNH
    if (body.action === "save_profile" || body.action === "save") {
      const convId = String(body.convId || "").trim();
      const customerName = String(body.customerName || "").trim();
      const profileText = String(body.data || body.profile || "").trim();

      if (!convId || convId === "default" || !profileText) {
        return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Missing convId or data" })).setMimeType(ContentService.MimeType.JSON);
      }

      const nowStr = saveConvProfileToSheet_(convId, customerName, profileText);
      return ContentService.createTextOutput(JSON.stringify({ success: true, time: nowStr })).setMimeType(ContentService.MimeType.JSON);
    }

    // 2. ĐỒNG BỘ KHÁCH GẮN NHÃN "MOBILE" TỪ PROJECT BDSD -> LỌC SERI & 5 TIN CUỐI
    if (body.action === "sync_mobile_tagged") {
      const result = handleSyncMobileTagged_(body);
      return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
    }

    // 3. FALLBACK RPC CHO WEB MOBILE (PHÒNG TRƯỜNG HỢP google.script.run BỊ CHẶN/LỖI)
    if (body.action === "mobile_rpc") {
      const fnName = String(body.fn || "").trim();
      const args = Array.isArray(body.args) ? body.args : [];
      const rpcMap = {
        mobileGetDashboardState: function(a) { return mobileGetDashboardState(a[0]); },
        mobileSyncWithBdsd: function() { return mobileSyncWithBdsd(); },
        mobileProcessPendingQueue: function() { return mobileProcessPendingQueue(); },
        mobileSaveSettings: function(a) { return mobileSaveSettings(a[0]); },
        mobileBuildContext: function(a) { return mobileBuildContext(a[0]); },
        mobileReadProfile: function(a) { return mobileReadProfile(a[0]); },
        mobileClearContext: function(a) { return mobileClearContext(a[0]); },
        mobileRunDivination: function(a) { return mobileRunDivination(a[0]); },
        mobileClearResultCache: function(a) { return mobileClearResultCache(a[0], a[1], a[2]); },
        mobileClearProfileCache: function(a) { return mobileClearProfileCache(a[0]); },
        mobileSaveDraft: function(a) { return mobileSaveDraft(a[0], a[1]); }
      };
      if (!rpcMap[fnName]) {
        return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Unknown RPC function: " + fnName })).setMimeType(ContentService.MimeType.JSON);
      }
      const rpcResult = rpcMap[fnName](args);
      return ContentService.createTextOutput(JSON.stringify(rpcResult)).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Invalid post action" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getMobileSettings_() {
  const store = PropertiesService.getScriptProperties();
  let saved = {};
  try {
    const raw = store.getProperty(MOB_PROP_KEYS.SETTINGS);
    if (raw) saved = JSON.parse(raw) || {};
  } catch (e) {}
  return Object.assign({}, MOB_DEFAULT_SETTINGS, saved);
}

function saveMobileSettings_(newSettings) {
  const store = PropertiesService.getScriptProperties();
  const current = getMobileSettings_();
  const merged = Object.assign({}, current, newSettings || {});
  store.setProperty(MOB_PROP_KEYS.SETTINGS, JSON.stringify(merged));
  return merged;
}

function getMobileSapoToken_() {
  return PropertiesService.getScriptProperties().getProperty(MOB_PROP_KEYS.SAPO_TOKEN) || '';
}

function getMobileTenant_() {
  return PropertiesService.getScriptProperties().getProperty(MOB_PROP_KEYS.SAPO_TENANT) || 'janet.mysapo.net';
}

function saveMobileSapoAuth_(token, tenant) {
  const store = PropertiesService.getScriptProperties();
  if (token) store.setProperty(MOB_PROP_KEYS.SAPO_TOKEN, String(token).trim());
  if (tenant) store.setProperty(MOB_PROP_KEYS.SAPO_TENANT, String(tenant).trim());
}

function removeMobileCustomerCaches_(convIds) {
  if (!Array.isArray(convIds) || convIds.length === 0) return;
  try {
    const cache = CacheService.getScriptCache();
    const keys = [];
    convIds.forEach(function(cid) {
      const clean = String(cid || '').trim();
      if (!clean) return;
      keys.push('draft_' + clean);
      keys.push('ctx_res_' + clean);
    });
    if (keys.length > 0) cache.removeAll(keys);
  } catch (e) {}
}

function readMobileCustIndex_() {
  const store = PropertiesService.getScriptProperties();
  let list = [];
  try {
    const raw = store.getProperty(MOB_PROP_KEYS.CUST_INDEX);
    if (raw) list = JSON.parse(raw);
    if (!Array.isArray(list)) list = [];
  } catch (e) {
    list = [];
  }
  const now = Date.now();
  const valid = [];
  const expiredIds = [];
  let changed = false;
  list.forEach(function(item) {
    if (!item || !item.convId) return;
    if (item.addedAt && (now - item.addedAt > MOB_TTL_MS)) {
      store.deleteProperty(MOB_PROP_KEYS.CUST_PREFIX + item.convId);
      expiredIds.push(item.convId);
      changed = true;
    } else {
      valid.push(item);
    }
  });
  if (changed) {
    store.setProperty(MOB_PROP_KEYS.CUST_INDEX, JSON.stringify(valid));
    removeMobileCustomerCaches_(expiredIds);
  }
  return valid;
}

function getMobileCustomer_(convId) {
  const raw = PropertiesService.getScriptProperties().getProperty(MOB_PROP_KEYS.CUST_PREFIX + convId);
  if (!raw) return null;
  try {
    return JSON.parse(raw);
  } catch (e) {
    return null;
  }
}

function saveMobileCustomer_(custObj) {
  if (!custObj || !custObj.convId) return;
  const store = PropertiesService.getScriptProperties();
  // Chỉ giữ metadata nhẹ trong PropertiesService (nháp dài & kết quả copy nằm ở CacheService 6h)
  const copy = Object.assign({}, custObj);
  delete copy.draft;
  delete copy.recentMessages;
  delete copy.detectedSerials;
  delete copy.initialScanned;
  delete copy.initialScanError;
  if (copy.ctxError && String(copy.ctxError).length > 300) {
    copy.ctxError = String(copy.ctxError).slice(0, 300);
  }
  store.setProperty(MOB_PROP_KEYS.CUST_PREFIX + copy.convId, JSON.stringify(copy));
}

function removeMobileCustomer_(convId) {
  const store = PropertiesService.getScriptProperties();
  store.deleteProperty(MOB_PROP_KEYS.CUST_PREFIX + convId);
  removeMobileCustomerCaches_([convId]);
  const indexList = readMobileCustIndex_().filter(function(it) { return it.convId !== convId; });
  store.setProperty(MOB_PROP_KEYS.CUST_INDEX, JSON.stringify(indexList));
}

function applyTaggedCustomerSyncList_(taggedCustomers, untaggedConvIds) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  const store = PropertiesService.getScriptProperties();
  try {
    let indexList = readMobileCustIndex_();
    const indexMap = {};
    indexList.forEach(function(it) { indexMap[it.convId] = it; });

    // 1. Xóa ngay khách vừa bị gỡ nhãn Mobile
    if (Array.isArray(untaggedConvIds)) {
      const removedIds = [];
      untaggedConvIds.forEach(function(cid) {
        const cleanId = String(cid || '').trim();
        if (!cleanId) return;
        store.deleteProperty(MOB_PROP_KEYS.CUST_PREFIX + cleanId);
        delete indexMap[cleanId];
        removedIds.push(cleanId);
      });
      removeMobileCustomerCaches_(removedIds);
    }

    // 2. Đồng bộ danh sách khách đang gắn nhãn Mobile (KHÔNG quét lại nếu khách đang treo chưa gỡ nhãn)
    const needInitialScanIds = [];
    const now = Date.now();

    if (Array.isArray(taggedCustomers)) {
      taggedCustomers.forEach(function(item) {
        const cid = String(item && item.convId || '').trim();
        if (!cid) return;

        const existing = getMobileCustomer_(cid);
        if (!existing) {
          // Khách MỚI phát hiện gắn nhãn (hoặc vừa gỡ nhãn rồi gắn lại):
          // Trạng thái mặc định 'idle' -> hiển thị nút "📋 Hồ sơ", KHÔNG tự gọi API AI
          const newCust = {
            convId: cid,
            pageId: String(item.pageId || ''),
            fbPageId: String(item.fbPageId || ''),
            customerName: String(item.customerName || 'Khách Hàng'),
            avatar: String(item.avatar || ''),
            taggedAt: Number(item.taggedAt || now),
            addedAt: now,
            ctxStatus: 'idle',
            ctxType: '',
            ctxTime: '',
            ctxError: '',
            unlocked: false,
            initialScanned: false,
            detectedSerials: [],
            recentMessages: []
          };
          saveMobileCustomer_(newCust);
          indexMap[cid] = { convId: cid, addedAt: now, taggedAt: newCust.taggedAt };
          needInitialScanIds.push(cid);
        } else {
          // Khách đang treo sẵn: chỉ cập nhật tên/avatar nếu đổi, KHÔNG quét lại tin nhắn
          let dirty = false;
          if (item.customerName && item.customerName !== existing.customerName) {
            existing.customerName = String(item.customerName);
            dirty = true;
          }
          if (item.avatar && item.avatar !== existing.avatar) {
            existing.avatar = String(item.avatar);
            dirty = true;
          }
          if (item.pageId && !existing.pageId) {
            existing.pageId = String(item.pageId);
            dirty = true;
          }
          if (dirty) saveMobileCustomer_(existing);
          if (!existing.initialScanned) needInitialScanIds.push(cid);
        }
      });
    }

    const updatedIndex = Object.keys(indexMap).map(function(k) { return indexMap[k]; });
    updatedIndex.sort(function(a, b) { return (b.taggedAt || b.addedAt || 0) - (a.taggedAt || a.addedAt || 0); });
    store.setProperty(MOB_PROP_KEYS.CUST_INDEX, JSON.stringify(updatedIndex));
    return needInitialScanIds;
  } finally {
    try { lock.releaseLock(); } catch (eLock) {}
  }
}

function handleSyncMobileTagged_(body) {
  if (body.sapoToken || body.tenant) {
    saveMobileSapoAuth_(body.sapoToken, body.tenant);
  }
  const taggedCustomers = Array.isArray(body.taggedCustomers) ? body.taggedCustomers : [];
  const untaggedConvIds = Array.isArray(body.untaggedConvIds) ? body.untaggedConvIds : [];

  const needInitialIds = applyTaggedCustomerSyncList_(taggedCustomers, untaggedConvIds);

  return {
    success: true,
    totalNewFound: needInitialIds.length,
    scannedCount: 0,
    remainingPending: 0
  };
}

function formatSapoMsgTime_(ts, onlyHm) {
  try {
    const d = new Date(Number(ts));
    if (isNaN(d.getTime())) return '';
    return Utilities.formatDate(d, 'Asia/Ho_Chi_Minh', onlyHm ? 'HH:mm' : 'dd/MM/yyyy HH:mm');
  } catch (e) {
    return '';
  }
}

function fetchSapoMessagesServer_(convId, token, tenant, maxPages) {
  if (!convId || convId === 'default') throw new Error('Thiếu mã hội thoại Sapo');
  if (!token) throw new Error('Chưa có Token Sapo (Hãy kết nối BDSD & mở Sapo trên PC)');

  const headers = {
    'Content-Type': 'application/json',
    'X-Bizweb-App-Fpage-Token': token,
    'X-Sapo-Tenant': tenant || 'janet.mysapo.net'
  };

  const allMsgs = [];
  const seenIds = {};
  let lastTime = null;
  const userMeta = { name: '', gender: '' };
  const pageCap = maxPages || MOB_MAX_PAGES;

  for (let page = 1; page <= pageCap; page++) {
    let url = 'https://sapo-socials.sapoapps.vn/api/conversations/' + encodeURIComponent(convId) + '/messages?language=vi';
    if (lastTime) url += '&last_time=' + encodeURIComponent(lastTime);

    const res = UrlFetchApp.fetch(url, {
      method: 'get',
      headers: headers,
      muteHttpExceptions: true
    });

    const code = res.getResponseCode();
    if (code < 200 || code >= 300) {
      throw new Error('Lỗi tải tin nhắn Sapo (HTTP ' + code + ')');
    }

    const data = JSON.parse(res.getContentText() || '{}');
    if (page === 1) {
      userMeta.name = (data && data.user && data.user.name) || (data && data.conversation && data.conversation.account_name) || '';
      userMeta.gender = (data && data.user && data.user.gender) || '';
    }

    const msgs = Array.isArray(data.messages) ? data.messages : [];
    if (msgs.length === 0) break;

    let added = 0;
    let oldestTs = Infinity;
    for (let i = 0; i < msgs.length; i++) {
      const m = msgs[i];
      const ts = Number(m.channel_created_on || 0);
      if (ts > 0 && ts < oldestTs) oldestTs = ts;
      const mid = String(m.id || ('idx_' + ts + '_' + i));
      if (!seenIds[mid]) {
        seenIds[mid] = true;
        allMsgs.push(m);
        added++;
      }
    }

    if (added === 0 || !isFinite(oldestTs)) break;
    lastTime = oldestTs;
  }

  allMsgs.sort(function(a, b) {
    return Number(a.channel_created_on || 0) - Number(b.channel_created_on || 0);
  });

  return { messages: allMsgs, userMeta: userMeta };
}

function extractCustomerRecentAndSerials_(messages) {
  const customerMsgs = [];
  for (let i = 0; i < messages.length; i++) {
    const m = messages[i];
    if (!m || m.type !== 'text' || m.is_page_message) continue;
    if (m.action === 'SEND_AUTOMATION_MESSAGE') continue;
    const txt = String(m.content || '').trim();
    if (!txt) continue;
    customerMsgs.push({
      id: String(m.id || ('m_' + i)),
      text: txt,
      time: formatSapoMsgTime_(m.channel_created_on, true)
    });
  }

  // Giữ 5 tin nhắn gần nhất của khách theo đúng trình tự thời gian để bấm ghép nối tiếp tự nhiên
  const recentMessages = customerMsgs.slice(-5);

  // Lọc số Seri bằng đúng Regex như Extension PC (quét từ tin mới nhất ngược lại để seri mới đứng đầu)
  const detectedSerials = [];
  const seenSerial = {};
  for (let i = customerMsgs.length - 1; i >= 0; i--) {
    const matches = customerMsgs[i].text.match(/[a-zA-Z0-9]{0,5}\s?\d{6,12}/gi);
    if (!matches) continue;
    for (let j = matches.length - 1; j >= 0; j--) {
      const digits = matches[j].replace(/\D/g, '');
      if (digits.length >= 6 && digits.length <= 12 && !seenSerial[digits]) {
        seenSerial[digits] = true;
        detectedSerials.push(digits);
      }
    }
    if (detectedSerials.length >= 6) break;
  }

  return { recentMessages: recentMessages, detectedSerials: detectedSerials };
}

function filterMessagesServer_(messages) {
  const filteredLines = [];
  for (let i = 0; i < messages.length; i++) {
    const m = messages[i];
    if (m.type !== 'text') continue;
    if (m.action === 'SEND_AUTOMATION_MESSAGE') continue;

    const txt = String(m.content || '').trim();
    if (!txt || txt.length <= 2) continue;

    const isShop = !!m.is_page_message;
    const lower = txt.toLowerCase();
    if (isShop) {
      let skip = false;
      for (let p = 0; p < MOB_SKIP_PATTERNS.length; p++) {
        if (lower.indexOf(MOB_SKIP_PATTERNS[p]) !== -1) {
          skip = true;
          break;
        }
      }
      if (skip) continue;
    }

    const dt = formatSapoMsgTime_(m.channel_created_on, false);
    const senderLabel = isShop ? 'THẦY (SHOP)' : 'KHÁCH';
    filteredLines.push((dt ? '[' + dt + '] ' : '') + senderLabel + ': ' + txt);
  }
  return filteredLines;
}

function callGeminiRawServer_(modelName, apiKey, payload) {
  if (!apiKey) throw new Error('Chưa cấu hình Gemini API Key');
  const mod = modelName || 'gemini-3.5-flash-lite';
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + encodeURIComponent(mod) + ':generateContent?key=' + encodeURIComponent(apiKey);

  const res = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });

  const code = res.getResponseCode();
  const raw = res.getContentText() || '';
  if (code < 200 || code >= 300) {
    let errMsg = 'HTTP ' + code;
    try {
      const errJson = JSON.parse(raw);
      if (errJson && errJson.error && errJson.error.message) errMsg = errJson.error.message;
    } catch (e) {}
    const err = new Error(errMsg);
    if (code === 429 || errMsg.toLowerCase().indexOf('quota') !== -1) err.isQuota = true;
    throw err;
  }
  return JSON.parse(raw);
}

function callOpenAICompatibleServer_(baseUrl, modelName, apiKey, userPrompt, systemPrompt, temperature, topP) {
  if (!apiKey) throw new Error('Chưa cấu hình API Key cho provider này');
  if (!modelName) throw new Error('Chưa cấu hình tên Model');
  if (!baseUrl) throw new Error('Chưa cấu hình Base URL');

  const cleanBase = String(baseUrl).replace(/\/+$/, '');
  const endpoint = cleanBase + '/chat/completions';

  const messages = [];
  if (systemPrompt) {
    messages.push({ role: 'system', content: systemPrompt });
  }
  messages.push({ role: 'user', content: userPrompt });

  const requestBody = {
    model: modelName,
    messages: messages
  };
  if (typeof temperature === 'number' && !isNaN(temperature)) {
    requestBody.temperature = temperature;
  }
  if (typeof topP === 'number' && !isNaN(topP)) {
    requestBody.top_p = topP;
  }
  if (cleanBase.indexOf('z.ai') !== -1 || cleanBase.indexOf('bigmodel') !== -1) {
    requestBody.thinking = { type: 'disabled' };
  }

  const res = UrlFetchApp.fetch(endpoint, {
    method: 'post',
    contentType: 'application/json',
    headers: {
      'Authorization': 'Bearer ' + apiKey,
      'Accept-Language': 'en-US,en'
    },
    payload: JSON.stringify(requestBody),
    muteHttpExceptions: true
  });

  const code = res.getResponseCode();
  const raw = res.getContentText() || '';
  if (code < 200 || code >= 300) {
    let errMsg = 'Error ' + code;
    try {
      const errJson = JSON.parse(raw);
      if (errJson && errJson.error && errJson.error.message) errMsg = errJson.error.message;
    } catch (e) {}
    throw new Error('[' + modelName + '] ' + errMsg);
  }

  const json = JSON.parse(raw);
  const text = (json && json.choices && json.choices[0] && json.choices[0].message && json.choices[0].message.content || '').trim();
  if (!text) throw new Error('[' + modelName + '] API trả về kết quả rỗng');
  return text;
}

    cache.remove(lockKey);
  }

function fetchTextWithCacheServer_(url, cacheKey) {
  const cache = CacheService.getScriptCache();
  try {
    const cached = cache.get(cacheKey);
    if (cached) return cached;
  } catch (e) {}

  try {
    const res = UrlFetchApp.fetch(url, { method: 'get', muteHttpExceptions: true });
    if (res.getResponseCode() >= 200 && res.getResponseCode() < 300) {
      const txt = res.getContentText() || '';
      if (txt) {
        // CacheService giới hạn ~100KB mỗi khóa; chỉ cache nếu an toàn < 90KB
        if (txt.length < 90000) {
          try { cache.put(cacheKey, txt, 21600); } catch (ePut) {}
        }
        return txt;
      }
    }
  } catch (eFetch) {}
  return '';
}

    if (dateObj && dateObj.year) {
      const p2 = function(n) { return String(n).padStart(2, '0'); };
      apiUrl += '&sa_date=' + dateObj.year + '-' + p2(dateObj.month) + '-' + p2(dateObj.day)
        + '&sa_hour=' + dateObj.hour + '&sa_minute=' + dateObj.min;
    }

    const lqRes = UrlFetchApp.fetch(apiUrl, { method: 'get', muteHttpExceptions: true });
    if (lqRes.getResponseCode() < 200 || lqRes.getResponseCode() >= 300) {
      throw new Error('Lỗi kết nối Lập quẻ (' + lqRes.getResponseCode() + ')');
    }
    const lapQueData = JSON.parse(lqRes.getContentText() || '{}');
    if (!lapQueData.success || !lapQueData.copyText) {
      throw new Error(lapQueData.error || 'Không thể lập quẻ.');
    }
    copyText = lapQueData.copyText;

    // 2. Tải kinh-dich.md & đọc Hồ sơ ngữ cảnh trên Sheet
    const mdContent = fetchTextWithCacheServer_(baseUrl + 'kinh-dich.md', 'CACHE_KINH_DICH_MD_V1');
    const historyContext = buildHistoryContextFromSheet_(convId);

    const systemPrompt = mdContent ? ('---\nKiến thức tham khảo:\n' + mdContent) : '';
    const userPrompt = copyText + (question ? (' ' + question) : '') + historyContext;

    const aiTemp = parseOptionalNumber_(settings.aiTemperature);
    const aiTopP = parseOptionalNumber_(settings.aiTopP);
    const rwTemp = parseOptionalNumber_(settings.rewriteTemperature);
    const rwTopP = parseOptionalNumber_(settings.rewriteTopP);

    const resCacheKey = 'res_' + convId + '_' + triggerSource + '_' + serial;

    try {
      let aiResult = '';
      let usedModelName = '';

      if (triggerSource === 'chu') {
        usedModelName = settings.customAiModel || 'Model Phụ';
        aiResult = callOpenAICompatibleServer_(
          settings.customAiBaseUrl,
          settings.customAiModel,
          settings.customAiApiKey,
          userPrompt,
          systemPrompt,
          aiTemp,
          aiTopP
        );
      } else {
        usedModelName = settings.geminiModel || 'gemini-3.5-flash-lite';
        if (settings.geminiBaseUrl && String(settings.geminiBaseUrl).trim()) {
          aiResult = callOpenAICompatibleServer_(
            settings.geminiBaseUrl,
            usedModelName,
            settings.geminiApiKey,
            userPrompt,
            systemPrompt,
            aiTemp,
            aiTopP
          );
        } else {
          const payload = {
            systemInstruction: { parts: [{ text: systemPrompt }] },
            contents: [{ parts: [{ text: userPrompt }] }]
          };
          const genConfig = {};
          if (aiTemp !== null) genConfig.temperature = aiTemp;
          if (aiTopP !== null) genConfig.topP = aiTopP;
          if (Object.keys(genConfig).length > 0) payload.generationConfig = genConfig;

          const json = callGeminiRawServer_(usedModelName, settings.geminiApiKey, payload);
          aiResult = (json && json.candidates && json.candidates[0] && json.candidates[0].content && json.candidates[0].content.parts && json.candidates[0].content.parts[0] && json.candidates[0].content.parts[0].text || '').trim();
        }
      }

      if (!aiResult || aiResult.length <= 10) {
        throw new Error('AI trả về kết quả rỗng');
      }

      // Cắt bỏ [TÓM_TẮT] nếu model lỡ tự viết thêm
      aiResult = aiResult.replace(/(?:\n+[-—~*_]*)*\s*(?:\n+\s*(?:[#*`_~-]*\s*)?\[?\s*TÓM[_ ]TẮT(?:\s+QUẺ)?\s*\]?\s*[*`_~-]*\s*:?[\s\S]*)$/i, '').trim();

      // BƯỚC 2: VIẾT LẠI THEO PHONG CÁCH THỰC TẾ (viet-lai.md)
      const vietLaiPrompt = fetchTextWithCacheServer_(baseUrl + 'viet-lai.md', 'CACHE_VIET_LAI_MD_V1');
      const rwModel = String(settings.rewriteModel || 'gemini-3.5-flash-lite').trim();
      const rwKey = String(settings.rewriteApiKey || '').trim() || (settings.rewriteType === 'openai_compatible' ? '' : String(settings.geminiApiKey || '').trim());

      let rewrittenResult = null;
      if (vietLaiPrompt && rwKey) {
        try {
          if (settings.rewriteType === 'openai_compatible' && settings.rewriteBaseUrl) {
            const textRw = callOpenAICompatibleServer_(
              settings.rewriteBaseUrl,
              rwModel,
              rwKey,
              aiResult,
              vietLaiPrompt,
              rwTemp,
              rwTopP
            );
            if (textRw && textRw.length > 20) rewrittenResult = textRw;
          } else {
            const rwPayload = {
              systemInstruction: { parts: [{ text: vietLaiPrompt }] },
              contents: [{ parts: [{ text: aiResult }] }]
            };
            const rwConfig = {};
            if (rwTemp !== null) rwConfig.temperature = rwTemp;
            if (rwTopP !== null) rwConfig.topP = rwTopP;
            if (Object.keys(rwConfig).length > 0) rwPayload.generationConfig = rwConfig;

            const rwJson = callGeminiRawServer_(rwModel, rwKey, rwPayload);
            const textRw = (rwJson && rwJson.candidates && rwJson.candidates[0] && rwJson.candidates[0].content && rwJson.candidates[0].content.parts && rwJson.candidates[0].content.parts[0] && rwJson.candidates[0].content.parts[0].text || '').trim();
            if (textRw && textRw.length > 20) rewrittenResult = textRw;
          }
        } catch (rwErr) {
          Logger.log('⚠️ Viết lại gặp sự cố, fallback bài gốc Bước 1: ' + rwErr.message);
        }
      }

      let resultObj;
      if (rewrittenResult) {
        resultObj = {
          success: true,
          type: 'business',
          content: rewrittenResult,
          rawContent: aiResult,
          model: rwModel,
          triggerSource: triggerSource,
          serial: serial,
          convId: convId,
          savedAt: Date.now()
        };
      } else {
        resultObj = {
          success: true,
          type: 'claude',
          content: aiResult,
          model: usedModelName,
          triggerSource: triggerSource,
          serial: serial,
          convId: convId,
          savedAt: Date.now()
        };
      }

      const serialized = JSON.stringify(resultObj);
      if (serialized.length < 95000) {
        try { cache.put(resCacheKey, serialized, 21600); } catch (eC) {}
      }
      return resultObj;
    } catch (aiErr) {
      // BƯỚC 1 LỖI -> FALLBACK QUẺ THÔ MÀU CAM (GEMINI)
      Logger.log('⚠️ Luận Bước 1 lỗi, fallback quẻ thô: ' + aiErr.message);
      const fallbackObj = {
        success: true,
        type: 'gemini',
        content: copyText,
        errorReason: aiErr && aiErr.message ? aiErr.message : '',
        triggerSource: triggerSource,
        serial: serial,
        convId: convId,
        savedAt: Date.now()
      };
      const serializedFb = JSON.stringify(fallbackObj);
      if (serializedFb.length < 95000) {
        try { cache.put(resCacheKey, serializedFb, 21600); } catch (eC2) {}
      }
      return fallbackObj;
    }
  } finally {
    cache.remove(runCacheKey);
  }
}

function getWebAppUrlSafe_() {
  const settings = getMobileSettings_();
  if (settings && settings.selfUrl && String(settings.selfUrl).indexOf('script.google.com') !== -1) {
    return String(settings.selfUrl).trim();
  }
  try {
    const svcUrl = ScriptApp.getService().getUrl();
    if (svcUrl && svcUrl.indexOf('script.google.com') !== -1) {
      return svcUrl;
    }
  } catch (e) {}
  return MOB_DEFAULT_SETTINGS.selfUrl;
}

function pullStateFromBdsd_() {
  const settings = getMobileSettings_();
  const bdsdUrl = String(settings.bdsdUrl || '').trim();
  if (!bdsdUrl) return null;

  try {
    const myUrl = getWebAppUrlSafe_();
    const res = UrlFetchApp.fetch(bdsdUrl, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify({
        action: myUrl ? 'register_context_url' : 'get_mobile_state',
        url: myUrl
      }),
      muteHttpExceptions: true
    });
    if (res.getResponseCode() >= 200 && res.getResponseCode() < 300) {
      const data = JSON.parse(res.getContentText() || '{}');
      if (data && data.success) {
        if (data.sapoToken || data.tenant) {
          saveMobileSapoAuth_(data.sapoToken, data.tenant);
        }
        return data;
      }
    }
  } catch (e) {
    Logger.log('⚠️ Lỗi kéo trạng thái từ BDSD: ' + e.message);
  }
  return null;
}

function mobileGetDashboardState(serialByConvMap) {
  const indexList = readMobileCustIndex_();
  const cache = CacheService.getScriptCache();
  const customers = [];

  // Đọc hàng loạt nháp (draft_*) và bản copy nhanh Hồ sơ (ctx_res_*) từ CacheService (6h)
  const draftHits = {};
  const ctxResHits = {};
  if (indexList.length > 0) {
    const batchKeys = [];
    indexList.forEach(function(idxItem) {
      batchKeys.push('draft_' + idxItem.convId);
      batchKeys.push('ctx_res_' + idxItem.convId);
    });
    try {
      const allBatch = cache.getAll(batchKeys);
      indexList.forEach(function(idxItem) {
        const cid = idxItem.convId;
        if (allBatch['draft_' + cid]) {
          try { draftHits[cid] = JSON.parse(allBatch['draft_' + cid]); } catch (eD) {}
        }
        if (allBatch['ctx_res_' + cid]) {
          try { ctxResHits[cid] = JSON.parse(allBatch['ctx_res_' + cid]); } catch (eC) {}
        }
      });
    } catch (eBatch) {}
  }

  indexList.forEach(function(idxItem) {
    const cust = getMobileCustomer_(idxItem.convId);
    if (!cust) return;

    if (draftHits[cust.convId]) {
      cust.draft = draftHits[cust.convId];
    }
    const ctxCache = ctxResHits[cust.convId];
    if (cust.ctxStatus === 'done' && cust.ctxType === 'ok' && ctxCache && ctxCache.content) {
      cust.hasCtxCache = true;
      cust.cachedProfileText = ctxCache.content;
    } else {
      cust.hasCtxCache = false;
    }

    // Kiểm tra trạng thái chạy AI và kết quả tạm đang nằm trong CacheService (khi thoát/mở lại trình duyệt)
    const activeRuns = {};
    const cachedResults = {};
    const serialHint = serialByConvMap && serialByConvMap[cust.convId] ? String(serialByConvMap[cust.convId]).trim() : '';
    const draftSerial = (cust.draft && cust.draft.serial) ? String(cust.draft.serial).trim() : '';
    const checkSerials = [];
    if (serialHint) checkSerials.push(serialHint);
    if (draftSerial && checkSerials.indexOf(draftSerial) === -1) checkSerials.push(draftSerial);

    if (checkSerials.length > 0) {
      const runKeys = checkSerials.map(function(s) { return 'run_' + cust.convId + '_' + s; });
      const resKeys = [];
      checkSerials.forEach(function(s) {
        resKeys.push('res_' + cust.convId + '_luan_' + s);
        resKeys.push('res_' + cust.convId + '_chu_' + s);
      });
      try {
        const runHits = cache.getAll(runKeys);
        Object.keys(runHits).forEach(function(k) {
          if (!runHits[k]) return;
          try {
            const parsed = JSON.parse(runHits[k]);
            if (parsed && parsed.startTime && (Date.now() - parsed.startTime < 240000)) {
              activeRuns[parsed.serial] = parsed;
            }
          } catch (eR) {}
        });
        const resHits = cache.getAll(resKeys);
        Object.keys(resHits).forEach(function(k) {
          if (!resHits[k]) return;
          try {
            const parsedRes = JSON.parse(resHits[k]);
            if (parsedRes && parsedRes.content) {
              cachedResults[parsedRes.triggerSource + '_' + parsedRes.serial] = parsedRes;
            }
          } catch (eRes) {}
        });
      } catch (eCache) {}
    }

    cust.activeRuns = activeRuns;
    cust.cachedResults = cachedResults;
    customers.push(cust);
  });

  const settings = getMobileSettings_();
  return {
    success: true,
    webAppUrl: getWebAppUrlSafe_(),
    hasSapoToken: !!getMobileSapoToken_(),
    settings: settings,
    customers: customers
  };
}

function mobileSyncWithBdsd() {
  const bdsdData = pullStateFromBdsd_();
  if (bdsdData && Array.isArray(bdsdData.taggedCustomers)) {
    const taggedMap = {};
    bdsdData.taggedCustomers.forEach(function(tc) {
      if (tc && tc.convId) taggedMap[String(tc.convId)] = true;
    });
    const toRemove = readMobileCustIndex_()
      .map(function(it) { return it.convId; })
      .filter(function(cid) { return !taggedMap[cid]; });
    applyTaggedCustomerSyncList_(bdsdData.taggedCustomers, toRemove);
  }
  return mobileGetDashboardState({});
}

function mobileSaveDraft(convId, draftPatch) {
  const cid = String(convId || '').trim();
  if (!cid || cid === 'default') return { success: false };
  if (!getMobileCustomer_(cid)) return { success: false };

  const cache = CacheService.getScriptCache();
  const cacheKey = 'draft_' + cid;
  let prev = {};
  try {
    const rawPrev = cache.get(cacheKey);
    if (rawPrev) prev = JSON.parse(rawPrev) || {};
  } catch (ePrev) {}

  const src = (draftPatch && typeof draftPatch === 'object') ? draftPatch : {};
  const draftObj = {
    serial: (src.serial !== undefined) ? String(src.serial || '').slice(0, 50) : String(prev.serial || '').slice(0, 50),
    question: (src.question !== undefined) ? String(src.question || '').slice(0, 45000) : String(prev.question || '').slice(0, 45000),
    dtVal: (src.dtVal !== undefined) ? String(src.dtVal || '').slice(0, 30) : String(prev.dtVal || '').slice(0, 30),
    tz: (src.tz !== undefined) ? String(src.tz || '').slice(0, 16) : String(prev.tz || '').slice(0, 16),
    savedAt: Number(src.savedAt || Date.now())
  };
  try {
    const serialized = JSON.stringify(draftObj);
    if (serialized.length < 95000) {
      cache.put(cacheKey, serialized, 21600);
    }
  } catch (ePut) {}
  return { success: true, draft: draftObj };
}
